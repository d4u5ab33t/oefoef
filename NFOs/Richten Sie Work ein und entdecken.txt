import pandas as pd
from pathlib import Path

class RoyaltyCalculator:
    def __init__(self, distributor_data_path):
        self.distributor_data = pd.read_csv(distributor_data_path)
        
    def calculate_splits(self, brutto, aggregator_fee=10, label_share=50):
        """Berechnet Splits basierend auf Vertrag"""
        net = brutto * (1 - aggregator_fee/100)
        artist_share = net * (1 - label_share/100)
        label_share = net * (label_share/100)
        return {
            'brutto': brutto,
            'netto': net,
            'kuenstler': artist_share,
            'label': label_share,
            'aggregator_gebuehr': brutto * aggregator_fee/100
        }
        
    def import_distributor_csv(self, file_path):
        """Importiert CSV von DistroKid/TuneCore"""
        data = pd.read_csv(file_path)
        # Automatische Zuordnung zu Künstlern/Releases
        return data

# Nutzung
calc = RoyaltyCalculator('/workspace/label-system/finance/distributor_data/')
result = calc.calculate_splits(1000, aggregator_fee=10, label_share=50)

import requests

class AIDemoScorer:
    def __init__(self, api_key):
        self.api_key = api_key
        
    def analyze_demo(self, audio_file_path):
        """Analysiert Demo mit KI"""
        with open(audio_file_path, 'rb') as f:
            files = {'audio': f}
            response = requests.post(
                'https://api.soundraw.io/analyze',
                files=files,
                headers={'Authorization': f'Bearer {self.api_key}'}
            )
        return response.json()
        
    def calculate_score(self, analysis):
        """Berechnet Gesamt-Score"""
        quality = analysis['quality_score'] * 0.4
        originality = analysis['originality_score'] * 0.3
        market_fit = analysis['market_fit_score'] * 0.2
        label_fit = self._calculate_label_fit(analysis) * 0.1
        return quality + originality + market_fit + label_fit

# Nutzung
scorer = AIDemoScorer('YOUR_API_KEY')
analysis = scorer.analyze_demo('/workspace/label-system/ar/inbox/demo.mp3')
score = scorer.calculate_score(analysis)
# Output: 7.8/10 → Automatische Empfehlung
