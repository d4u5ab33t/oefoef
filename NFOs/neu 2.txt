def _agent_key(s: str) -> str:

def run_trend_agent_network(root: Path, out: Path, enabled=True) -> dict[str, Any]:
    """
    Run oida_trend_agents.py and return a compact, track-matchable snapshot.

    The agent network remains a separate process/module so the viral grid can
    still run offline if the network is unavailable.
    """
    state_file = out / "trend_agent_state.json"

    agent_script = Path(__file__).with_name("oida_trend_agents.py")
    if not agent_script.exists():
        # Also support running the two scripts from the same output/tool dir.
        alt = out.parent / "oida_trend_agents.py"

            "status": "missing_agent_script",
            "agents": {},
            "network": {},
            "error": f"Nicht gefunden: {agent_script}",
        }

    try:
        cmd = [
            sys.executable, str(agent_script),
            "--root", str(root),
            "--out", str(out),
        ]
        proc = subprocess.run(
            cmd, capture_output=True, text=True, timeout=900,
            encoding="utf-8", errors="replace"
        )
        if proc.returncode != 0:
            return {
                "status": "agent_error",
                "agents": {},
                "network": {},
                "error": (proc.stderr or proc.stdout)[-4000:],
            }

            "network": {},

def _track_text(t: Track) -> str:
    return norm(" ".join([
        t.title, t.artist, t.genre, t.comment, t.lyrics,
        str(t.key), str(t.bpm), str(t.release_date)
    ]))

def _term_match(text: str, term: str) -> bool:
    term = norm(term)
    if not term:
        return False
    return term in text

def score_track_against_agent(t: Track, agent: dict[str, Any]) -> dict[str, Any]:
    """
    Match a track against an agent's observed trend vocabulary.

    Score = metadata/text fit × observed trend pressure × confidence.
    This intentionally stays explainable: every boost has a matching term.
    """
    cfg_terms = agent.get("_terms", {})
    top_terms = agent.get("top_terms", {}) or {}
    signal_scores = agent.get("signal_scores", {}) or {}
    text = _track_text(t)

    hits = []
    raw = 0.0
    for term, weight in cfg_terms.items():
        if _term_match(text, term):
            observed = min(1.0, float(top_terms.get(term, 0)) / 5.0)
            boost = 0.55 + 0.45 * observed
            contribution = float(weight) * boost
            raw += contribution
            hits.append({
                "term": term,
                "weight": round(float(weight), 3),
                "observed": round(observed, 3),
                "contribution": round(contribution, 3),
            })

    # Signal names are often broader than exact metadata terms.
    signal_bonus = 0.0
    for sig, val in signal_scores.items():
        if _term_match(text, sig):
            signal_bonus += 0.10 * float(val)

    raw += signal_bonus
    denom = max(2.0, sum(float(x) for x in cfg_terms.values()) * 0.30)
    fit = max(0.0, min(1.0, raw / denom))

    confidence = float(agent.get("confidence", 0.5) or 0.5)
    trend = float(agent.get("trend_score", 0.0) or 0.0)
    forecast30 = float((agent.get("forecast") or {}).get("30d", trend) or trend)

    # Agent impact: local track fit + current regional momentum.
    momentum = 0.55 * trend + 0.45 * forecast30
    impact = max(0.0, min(1.0, fit * (0.65 + 0.35 * confidence) *
                              (0.70 + 0.30 * momentum)))

    return {
        "fit": round(fit, 4),
        "impact": round(impact, 4),
        "confidence": round(confidence, 4),
        "trend": round(trend, 4),
        "forecast30": round(forecast30, 4),
        "hits": sorted(hits, key=lambda x: -x["contribution"])[:8],
    }

def enrich_agent_reports(agent_net: dict[str, Any]) -> dict[str, Any]:
    """
    Attach the term dictionaries from the agent script to the persisted
    reports. This makes the state self-contained for the grid.
    """
    # Kept in sync with the dedicated agent network. If that module changes,
    # import it rather than duplicating the dictionaries here.
    try:
        import importlib.util
        agent_path = Path(__file__).with_name("oida_trend_agents.py")
        if not agent_path.exists():
            agent_path = Path("oida_trend_agents.py")
        spec = importlib.util.spec_from_file_location("oida_trend_agents_local", agent_path)
        if spec and spec.loader:
            mod = importlib.util.module_from_spec(spec)
            spec.loader.exec_module(mod)
            for aid, report in agent_net.get("agents", {}).items():
                if aid in getattr(mod, "AGENTS", {}):
                    report["_terms"] = mod.AGENTS[aid].get("terms", {})
                    report["_weight"] = mod.AGENTS[aid].get("weight", 1.0)
    except Exception:
        pass
    return agent_net

def apply_agent_scores(tracks: list[Track], agent_net: dict[str, Any]) -> None:
    """
    Fuse the 10-agent network into each Track's trend_score and viral_score.

    Important: the agent layer is a bounded component, not a replacement for
    the existing local/audio/vote signals.
    """
    agents = agent_net.get("agents", {})
    if not agents:
        return

    ranked_network = []
    weighted_impacts = []
    for t in tracks:
        per_agent = {}
        weighted = 0.0
        total_w = 0.0

        for aid in AGENT_IDS:
            a = agents.get(aid)
            if not a:
                continue
            result = score_track_against_agent(t, a)
            per_agent[aid] = result

            # Agent's configured regional weight.
            w = float(a.get("_weight", 1.0) or 1.0)
            weighted += result["impact"] * w
            total_w += w

        agent_score = weighted / total_w if total_w else 0.0

        # Preserve the original score and add a bounded agent contribution.
        # 35% max prevents web noise from overwhelming actual track evidence.
        old_trend = float(t.trend_score or 0.0)
        fused_trend = max(0.0, min(1.0, 0.65 * old_trend + 0.35 * agent_score))
        t.trend_score = round(fused_trend, 4)

        # Existing viral_score is 0..100. Add at most 12.0 points from agents.
        old_viral = float(t.viral_score or 0.0)
        delta = (agent_score - 0.5) * 24.0
        t.viral_score = round(max(0.0, min(100.0, old_viral + delta)), 1)

        ordered = sorted(per_agent.items(), key=lambda kv: -kv[1]["impact"])
        t.agent_scores = per_agent
        t.agent_score = round(agent_score, 4)
        t.best_agents = [a for a, _ in ordered[:4]]
        t.agent_reason = "; ".join(
            f"{aid} {res['impact']:.2f}"
            + (f" [{', '.join(h['term'] for h in res['hits'][:3])}]" if res["hits"] else "")
            for aid, res in ordered[:4]
        )

        ranked_network.append((t.viral_score, t))
        weighted_impacts.append(agent_score)

    if weighted_impacts:
        agent_net["mean_track_agent_score"] = round(
            sum(weighted_impacts) / len(weighted_impacts), 4
        )
        agent_net["tracks_scored"] = len(weighted_impacts)

        label = "BREAKOUT"
    elif a >= .62:
        label = "HIGH"
    elif a >= .46:
        label = "WATCH"
    else:
        label = "NICHE"

    if top:
        t.forecast = label
        t.forecast_reason = (
            f"Agent-Network: {label} ({a:.2f}); "
            f"stärkste Märkte: {', '.join(top)}. "
            + getattr(t, "forecast_reason", "")
        )
