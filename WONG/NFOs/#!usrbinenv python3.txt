import os
import shutil
import subprocess
import textwrap
from pathlib import Path

PROJECT_NAME = "weedit-native"
INSTALL_DIR = Path.cwd() / PROJECT_NAME


def write_file(path: str, content: str):
    """Schreibt Datei mit korrekter Einrückung."""
    p = Path(path)
    p.parent.mkdir(parents=True, exist_ok=True)
    clean_content = textwrap.dedent(content).strip() + "\n"
    with open(p, "w", encoding="utf-8") as f:
        f.write(clean_content)
    print(f"  ✅ {path}")


def kill_weed_processes():
    """Killt alle laufenden weed.exe Prozesse."""
    print("🔪 Killing old weed processes...")
    try:
        subprocess.run(
            ["taskkill", "/F", "/IM", "weed.exe"],
            capture_output=True, text=True
        )
        subprocess.run(
            ["taskkill", "/F", "/IM", "python.exe", "/FI", "WINDOWTITLE eq weed*"],
            capture_output=True, text=True
        )
    except Exception:
        pass


def uninstall_old():
    """Deinstalliert alte Version."""
    print("🗑️  Uninstalling old weedit-native...")
    try:
        subprocess.run(
            ["pip", "uninstall", "-y", "weedit-native"],
            capture_output=True, text=True
        )
    except Exception:
        pass
    
    if INSTALL_DIR.exists():
        print(f"  🧹 Removing {INSTALL_DIR}...")
        shutil.rmtree(INSTALL_DIR, ignore_errors=True)


def install_requirements():
    """Installiert Dependencies."""
    print("📦 Installing requirements...")
    req_file = INSTALL_DIR / "requirements.txt"
    try:
        subprocess.run(
            ["pip", "install", "-r", str(req_file)],
            check=True
        )
        print("  ✅ Requirements installed!")
    except Exception as e:
        print(f"  ⚠️  Warning: {e}")


def create_weed_bat():
    """Erstellt weed.bat Wrapper (umgeht WinError 32)."""
    print("🔧 Creating weed.bat wrapper...")
    
    bat_content = f"""@echo off
set PYTHONPATH={INSTALL_DIR}
python "{INSTALL_DIR}\\cli\\main.py" %*
"""
    
    # Schreibe weed.bat in das Projekt-Verzeichnis
    bat_path = INSTALL_DIR / "weed.bat"
    with open(bat_path, "w", encoding="utf-8") as f:
        f.write(bat_content)
    
    # Schreibe weed.bat auch in Python Scripts Ordner (für globalen Zugriff)
    try:
        result = subprocess.run(
            ["python", "-c", "import sysconfig; print(sysconfig.get_path('scripts'))"],
            capture_output=True, text=True, check=True
        )
        scripts_dir = Path(result.stdout.strip())
        if scripts_dir.exists():
            global_bat = scripts_dir / "weed.bat"
            with open(global_bat, "w", encoding="utf-8") as f:
                f.write(bat_content)
            print(f"  ✅ Global weed.bat created: {global_bat}")
    except Exception as e:
        print(f"  ⚠️  Could not create global weed.bat: {e}")
    
    print(f"  ✅ Local weed.bat created: {bat_path}")


def create_weed_ps1():
    """Erstellt weed.ps1 für PowerShell."""
    print("🔧 Creating weed.ps1 wrapper...")
    
    ps1_content = f"""$env:PYTHONPATH = "{INSTALL_DIR}"
python "{INSTALL_DIR}\\cli\\main.py" $args
"""
    
    ps1_path = INSTALL_DIR / "weed.ps1"
    with open(ps1_path, "w", encoding="utf-8") as f:
        f.write(ps1_content)
    
    # Auch global
    try:
        result = subprocess.run(
            ["python", "-c", "import sysconfig; print(sysconfig.get_path('scripts'))"],
            capture_output=True, text=True, check=True
        )
        scripts_dir = Path(result.stdout.strip())
        if scripts_dir.exists():
            global_ps1 = scripts_dir / "weed.ps1"
            with open(global_ps1, "w", encoding="utf-8") as f:
                f.write(ps1_content)
            print(f"  ✅ Global weed.ps1 created: {global_ps1}")

    print(f"  ✅ Local weed.ps1 created: {ps1_path}")

        import re
        from rich.console import Console

        console = Console()

        class SmartRenamer:
            def __init__(self, output_dir: Path):
                self.output_dir = output_dir
                self.output_dir.mkdir(parents=True, exist_ok=True)
            
            def get_next_version(self, base_name: str, platform: str = "youtube", extension: str = ".mp4") -> Path:
                clean_base = re.sub(r'_(youtube|tiktok|instagram|shorts|reels)_v\\d{3}$', '', base_name)

                for file in self.output_dir.iterdir():
                    match = pattern.match(file.name)
                    if match:

                next_version = max(existing_versions) + 1 if existing_versions else 1
                version_str = f"v{next_version:03d}"
                filename = f"{clean_base}_{platform_tag}_{version_str}{extension}"
                
                full_path = self.output_dir / filename

        import json
        import numpy as np
        from pathlib import Path
        from typing import List, Dict, Tuple
        from rich.console import Console

        console = Console()

        class DiversityTracker:
            def __init__(self, db_path: Path = Path(".weed_cache/diversity.db")):

                self.db_path.parent.mkdir(parents=True, exist_ok=True)

            def _init_db(self):
                self.conn.execute(\"\"\"CREATE TABLE IF NOT EXISTS usage (
                    segment_id TEXT PRIMARY KEY, clip_id TEXT,
                    usage_count INTEGER DEFAULT 0, last_used_in_shot TEXT,
                    total_reward REAL DEFAULT 0.0)\"\"\")

            def record_usage(self, segment_id: str, clip_id: str, shot_id: str, reward: float):
                cursor = self.conn.execute("SELECT usage_count, total_reward FROM usage WHERE segment_id = ?", (segment_id,))
                row = cursor.fetchone()
                if row:
                    self.conn.execute("UPDATE usage SET usage_count = ?, last_used_in_shot = ?, total_reward = ? WHERE segment_id = ?",
                        (row[0] + 1, shot_id, row[1] + reward, segment_id))
                else:
                    self.conn.execute("INSERT INTO usage (segment_id, clip_id, usage_count, last_used_in_shot, total_reward) VALUES (?, ?, 1, ?, ?)",
                        (segment_id, clip_id, shot_id, reward))
                self.conn.commit()
                self.recent_segments.append(segment_id)
                if len(self.recent_segments) > self.cooldown_size:
                    self.recent_segments.pop(0)
            
            def store_visual_features(self, segment_id: str, features: Dict):
                self.conn.execute(\"\"\"INSERT OR REPLACE INTO visual_features 
                    (segment_id, scene_type, energy_peak, motion_direction, emotion, color_dominance, visual_motifs)
                    VALUES (?, ?, ?, ?, ?, ?, ?)\"\"\", (

                    features.get("energy_peak", 0.5), features.get("motion_direction", "static"),
                    features.get("emotion", "neutral"), json.dumps(features.get("color_dominance", [])),
                    json.dumps(features.get("visual_motifs", []))))
                self.conn.commit()
            
            def is_in_cooldown(self, segment_id: str) -> bool:
                return segment_id in self.recent_segments
            
            def get_novelty_penalty(self, segment_id: str) -> float:
                cursor = self.conn.execute("SELECT usage_count FROM usage WHERE segment_id = ?", (segment_id,))
                row = cursor.fetchone()
                usage_count = row[0] if row else 0
                penalties = {0: 0.0, 1: 0.05, 2: 0.1, 3: 0.2, 4: 0.3}
                return penalties.get(usage_count, 0.4)
            
            def calculate_diversity_score(self, segment_id: str, recent_segments: List[str]) -> float:
                if not recent_segments:
                    return 1.0
                cursor = self.conn.execute("SELECT scene_type, energy_peak, motion_direction, emotion, color_dominance FROM visual_features WHERE segment_id = ?", (segment_id,))
                row = cursor.fetchone()
                if not row:
                    return 0.5
                curr_scene, curr_energy, curr_motion, curr_emotion, curr_colors_json = row
                curr_colors = json.loads(curr_colors_json) if curr_colors_json else []
                
                diversity_scores = []
                for recent_id in recent_segments[-5:]:
                    cursor = self.conn.execute("SELECT scene_type, energy_peak, motion_direction, emotion, color_dominance FROM visual_features WHERE segment_id = ?", (recent_id,))
                    recent_row = cursor.fetchone()
                    if not recent_row:
                        continue
                    rec_scene, rec_energy, rec_motion, rec_emotion, rec_colors_json = recent_row
                    rec_colors = json.loads(rec_colors_json) if rec_colors_json else []
                    
                    diffs = []
                    diffs.append(1.0 if curr_scene != rec_scene else 0.0)
                    diffs.append(abs(curr_energy - rec_energy))
                    diffs.append(1.0 if curr_motion != rec_motion else 0.0)
                    diffs.append(1.0 if curr_emotion != rec_emotion else 0.0)
                    
                    matches = sum(1 for c1 in curr_colors if c1 in rec_colors)
                    color_sim = matches / max(len(curr_colors), len(rec_colors), 1)
                    diffs.append(1.0 - color_sim)
                    
                    diversity_scores.append(np.mean(diffs))
                
                return float(np.mean(diversity_scores)) if diversity_scores else 0.5
            
            def get_statistics(self) -> Dict:
                cursor = self.conn.execute(\"\"\"SELECT 
                    COUNT(*) as total, SUM(CASE WHEN usage_count = 0 THEN 1 ELSE 0 END) as unused,
                    SUM(CASE WHEN usage_count = 1 THEN 1 ELSE 0 END) as used_once,
                    SUM(CASE WHEN usage_count > 1 THEN 1 ELSE 0 END) as used_multiple,
                    AVG(usage_count) as avg_usage, MAX(usage_count) as max_usage
                    FROM usage\"\"\")
                row = cursor.fetchone()
                return {"total_segments": row[0] or 0, "unused": row[1] or 0, "used_once": row[2] or 0,
                        "used_multiple": row[3] or 0, "avg_usage": row[4] or 0.0, "max_usage": row[5] or 0}
            
            def get_most_used_segments(self, limit: int = 10) -> List[Tuple[str, int]]:
                cursor = self.conn.execute("SELECT segment_id, usage_count FROM usage ORDER BY usage_count DESC LIMIT ?", (limit,))
                return [(row[0], row[1]) for row in cursor.fetchall()]
    """)

        class SongCompiler:
            def compile(self, mp3_path: Path) -> SongAST:
                y, sr = librosa.load(str(mp3_path), sr=None)
                duration_ms = int(len(y) / sr * 1000)
                tempo, beats = librosa.beat.beat_track(y=y, sr=sr)

                energy_curve = []
                for i in range(100):
                    t = (i / 99) * (duration_ms / 1000)
                    idx = np.argmin(np.abs(rms_times - t))
                    energy_curve.append(float(rms[idx]))

                beat_objs = [BeatAST(time_ms=bt, is_downbeat=(i % 4 == 0), energy=energy_curve[min(99, int((bt / duration_ms) * 100))]) for i, bt in enumerate(beat_times_ms)]
                section_defs = [("intro", 0.0, 0.15), ("verse", 0.15, 0.4), ("chorus", 0.4, 0.7), ("bridge", 0.7, 0.85), ("outro", 0.85, 1.0)]

                    sec = SectionAST(name=name, start_ms=start_ms, end_ms=end_ms, phrases=[phrase], bpm=tempo, dynamic_range=energy_level)
                    sections.append(sec)
                song_ast = SongAST(
                    title=mp3_path.stem.split(" - ")[-1] if " - " in mp3_path.stem else mp3_path.stem,
                    artist=mp3_path.stem.split(" - ")[0] if " - " in mp3_path.stem else "Unknown",
                    total_duration_ms=duration_ms, sections=sections,
                    global_energy_curve=energy_curve, global_emotion_curve=energy_curve)
                song_ast.metadata["genre"] = genre_name
                song_ast.metadata["genre_template"] = genre_template.name

                        novelty_penalty = self.diversity_tracker.get_novelty_penalty(segment_id)
                        score -= novelty_penalty
                        diversity_score = self.diversity_tracker.calculate_diversity_score(segment_id, self.diversity_tracker.recent_segments)
                        score += diversity_score * 0.2
                        candidates.append(ResolverCandidate(
                            clip_dna_id=cid, trim_start_ms=segment.start_ms, trim_end_ms=segment.end_ms,
                            constraint_match_score=max(0.0, min(1.0, score)), semantic_similarity=diversity_score))
                candidates.sort(key=lambda x: x.constraint_match_score, reverse=True)
                top_50_percent = max(1, len(candidates) // 2)
                top_candidates = candidates[:top_50_percent]
                rest_candidates = candidates[top_50_percent:]
                random.shuffle(rest_candidates)
                return ResolverResult(intent_id=intent.id, candidates=(top_candidates + rest_candidates)[:top_k], query_constraints={})
            
            def mark_clip_used(self, clip_id: str):
                self.used_clips.add(clip_id)
            
            def mark_segment_used(self, segment_id: str, clip_id: str, shot_id: str, reward: float):
                self.diversity_tracker.record_usage(segment_id, clip_id, shot_id, reward)

        class RLBandit:

            def choose(self, intent: ShotIntent, resolver_result: ResolverResult, song_ast: SongAST, prev_shots: List[RenderIR]) -> BanditDecision:

                    return base_score
                best = max(candidates, key=score_candidate)
                reward = best.constraint_match_score * 0.7 + best.semantic_similarity * 0.3
                return BanditDecision(intent_id=intent.id, chosen_candidate=best, rejected_candidates_count=len(candidates)-1, reward_prediction=reward, decision_reason=f"Maximized constraint match ({best.constraint_match_score:.2f}) + novelty bonus")

        import math
        from typing import Dict
        from core.ir import SongAST

        class VirtualCamera:
            def __init__(self, song_ast: SongAST, base_resolution: tuple = (1920, 1080)):
                self.song_ast = song_ast
                self.width, self.height = base_resolution
                self.max_zoom = 1.4
                self.min_zoom = 1.0
                self.max_pan = 0.15

            def get_transform(self, time_ms: int, section_name: str, energy: float) -> Dict[str, float]:
                total_duration = max(1, self.song_ast.total_duration_ms)
                t_norm = time_ms / total_duration
                base_zoom_speed = 0.2
                if section_name == "chorus": base_zoom_speed = 0.6
                elif section_name == "outro": base_zoom_speed = -0.3
                zoom = self.min_zoom + (self.max_zoom - self.min_zoom) * (t_norm * base_zoom_speed + energy * 0.2)
                zoom = max(self.min_zoom, min(self.max_zoom, zoom))
                freq_x = 1.5 if section_name in ["verse", "intro"] else 3.0
                pan_x = math.sin(t_norm * math.pi * freq_x) * self.max_pan * (1.0 - energy * 0.5)
                freq_y = 2.0
                pan_y = math.cos(t_norm * math.pi * freq_y) * (self.max_pan * 0.5)
                rotation = math.sin(t_norm * math.pi * 4) * (energy * 1.5)
                return {"zoom": round(zoom, 4), "x": round(pan_x, 4), "y": round(pan_y, 4), "rotation": round(rotation, 4)}
    """)

        from typing import List, Dict
        from core.ir import ShotIntent, ResolverResult, BanditDecision, RenderIR, SongAST, ReplayLog, ClipDNA
        from core.camera.virtual_camera import VirtualCamera
        from datetime import datetime

        class BeatAlignmentPass:
            def run(self, intent, candidate, current_ir: RenderIR, song_ast: SongAST, clip_dna: ClipDNA = None) -> RenderIR:
                snapped_in = (current_ir.trim_in_ms // 500) * 500
                return current_ir.model_copy(update={"trim_in_ms": snapped_in, "metadata": {**current_ir.metadata, "pass": "beat"}})

        class MotionFlowPass:
            def __init__(self, clip_dna_map: Dict[str, ClipDNA]):
                self.clip_dna_map = clip_dna_map
            def run(self, intent, candidate, current_ir: RenderIR, prev_shots: List[RenderIR]) -> RenderIR:
                current_clip = self.clip_dna_map.get(candidate.clip_dna_id)
                if not current_clip or not current_clip.motion_vectors: return current_ir
                updates = {}
                if prev_shots:
                    prev_shot = prev_shots[-1]
                    prev_clip_id = None
                    for clip_id, clip in self.clip_dna_map.items():
                        if clip.file_path == prev_shot.clip_file_path:
                            prev_clip_id = clip_id
                            break
                    if prev_clip_id and prev_clip_id in self.clip_dna_map:
                        prev_clip = self.clip_dna_map[prev_clip_id]
                        if prev_clip.motion_vectors:
                            prev_end = prev_clip.motion_vectors.get("end", {})
                            prev_end_speed = prev_end.get("speed", 0)
                            curr_start = current_clip.motion_vectors.get("start", {})
                            curr_start_speed = curr_start.get("speed", 0)
                            speed_diff = abs(prev_end_speed - curr_start_speed)
                            if speed_diff > 2.0:
                                ramp_factor = min(0.7, curr_start_speed / max(0.1, prev_end_speed))
                                updates["speed_ramp_in"] = ramp_factor
                            updates["motion_continuity"] = {"prev_speed": prev_end_speed, "curr_speed": curr_start_speed, "speed_ratio": curr_start_speed / max(0.1, prev_end_speed)}
                curr_end = current_clip.motion_vectors.get("end", {})
                curr_end_speed = curr_end.get("speed", 0)
                if curr_end_speed > 3.0: updates["speed_ramp_out"] = 1.3
                if updates: return current_ir.model_copy(update=updates)
                return current_ir

        class VirtualCameraPass:
            def __init__(self, virtual_camera: VirtualCamera):
                self.vcam = virtual_camera
            def run(self, intent: ShotIntent, candidate, current_ir: RenderIR, prev_shots: List[RenderIR]) -> RenderIR:
                energy = intent.need_energy
                start_transform = self.vcam.get_transform(time_ms=intent.start_ms, section_name=intent.song_section, energy=energy)
                end_transform = self.vcam.get_transform(time_ms=intent.end_ms, section_name=intent.song_section, energy=energy)
                return current_ir.model_copy(update={"camera_transform_start": start_transform, "camera_transform_end": end_transform})

        class OptimizerPipeline:
            def __init__(self, song_ast: SongAST, clip_dna_map: Dict[str, ClipDNA] = None):
                self.song_ast = song_ast
                self.clip_dna_map = clip_dna_map or {}
                self.virtual_camera = VirtualCamera(song_ast)
                self.passes = [BeatAlignmentPass(), MotionFlowPass(self.clip_dna_map), VirtualCameraPass(self.virtual_camera)]
                self.replay_logs = []
                
            def optimize_all(self, intents: List[ShotIntent], resolver_results: Dict, bandit_decisions: Dict, initial_irs: Dict) -> List[RenderIR]:
                timeline, prev_shots = [], []
                for intent in intents:
                    ir = initial_irs[intent.id]
                    applied = []
                    candidate = bandit_decisions[intent.id].chosen_candidate
                    clip_dna = self.clip_dna_map.get(candidate.clip_dna_id)
                    for p in self.passes:
                        if isinstance(p, BeatAlignmentPass): ir = p.run(intent, candidate, ir, self.song_ast, clip_dna)
                        elif isinstance(p, MotionFlowPass): ir = p.run(intent, candidate, ir, prev_shots)
                        elif isinstance(p, VirtualCameraPass): ir = p.run(intent, candidate, ir, prev_shots)
                        applied.append(p.__class__.__name__)
                    timeline.append(ir)
                    prev_shots.append(ir)
                    self.replay_logs.append(ReplayLog(
                        shot_id=intent.id, timestamp=datetime.utcnow(), intent=intent,
                        resolver_result=resolver_results[intent.id], bandit_decision=bandit_decisions[intent.id],
                        final_reward=bandit_decisions[intent.id].reward_prediction, optimizer_passes_applied=applied))
                return timeline
    """)

        import subprocess
        import json

        from typing import List, Tuple

        def get_video_dimensions(video_path: Path) -> Tuple[int, int]:
            try:
                cmd = ["ffprobe", "-v", "quiet", "-print_format", "json", "-show_streams", str(video_path)]

                    out_clip = temp_dir / f"clip_{i:04d}.mp4"

                    filters = [f"trim=start={start_sec}:duration={duration_sec}", "setpts=PTS-STARTPTS"]

                        ramp_duration = min(0.5, duration_sec / 3)
                        filters.append(f"setpts='if(lt(T\\,{ramp_duration})\\,PTS/{ir.speed_ramp_in}\\,PTS)'")
                    if ir.speed_ramp_out and ir.speed_ramp_out != 1.0:
                        ramp_start = max(0, duration_sec - 0.5)
                        filters.append(f"setpts='if(gt(T\\,{ramp_start})\\,PTS*{ir.speed_ramp_out}\\,PTS)'")
                    
                    zoom_to_cover = max(target_width / orig_w, target_height / orig_h)
                    
                    vcam_zoom = 1.0
                    vcam_x = 0.0
                    vcam_y = 0.0
                    
                    if ir.camera_transform_start and ir.camera_transform_end:
                        t_start = ir.camera_transform_start
                        t_end = ir.camera_transform_end
                        vcam_zoom = (t_start["zoom"] + t_end["zoom"]) / 2.0
                        vcam_x = (t_start["x"] + t_end["x"]) / 2.0
                        vcam_y = (t_start["y"] + t_end["y"]) / 2.0
                    
                    total_zoom = zoom_to_cover * vcam_zoom
                    filters.append(f"scale=iw*{total_zoom}:ih*{total_zoom}")
                    
                    scaled_w = orig_w * total_zoom
                    scaled_h = orig_h * total_zoom
                    
                    if scaled_w / scaled_h > target_ratio:
                        crop_w = scaled_h * target_ratio
                        crop_h = scaled_h
                    else:
                        crop_w = scaled_w
                        crop_h = scaled_w / target_ratio
                    
                    max_offset_x = (scaled_w - crop_w) / 2.0
                    max_offset_y = (scaled_h - crop_h) / 2.0
                    
                    crop_x = (scaled_w - crop_w) / 2.0 + (vcam_x * max_offset_x * 2)
                    crop_y = (scaled_h - crop_h) / 2.0 + (vcam_y * max_offset_y * 2)
                    
                    crop_x = max(0, min(crop_x, scaled_w - crop_w))
                    crop_y = max(0, min(crop_y, scaled_h - crop_h))
                    
                    filters.append(f"crop={int(crop_w)}:{int(crop_h)}:{int(crop_x)}:{int(crop_y)}")
                    filters.append(f"scale={target_width}:{target_height}")

                    vf_string = ",".join(filters)
                    
                    cmd = ["ffmpeg", "-y", "-i", str(ir.clip_file_path), "-vf", vf_string,
                           "-an", "-c:v", codec, "-b:v", video_bitrate, "-r", str(target_fps), 
                           "-preset", "fast", "-pix_fmt", "yuv420p", str(out_clip)]
                    
                    try:
                        result = subprocess.run(cmd, check=True, capture_output=True, text=True)
                        processed.append(out_clip)

                    except subprocess.CalledProcessError as e:
                        print(f"\\n[red]❌ FFmpeg failed for {Path(ir.clip_file_path).name}[/red]")

                    except Exception as e:

                if not processed:
                    raise RuntimeError("No clips processed. Check FFmpeg errors above.")
                    
                concat_file = temp_dir / "concat.txt"
                with open(concat_file, "w", encoding="utf-8") as f:
                    for p in processed: 
                        f.write(f"file '{p.as_posix()}'\\n")

                for p in processed: p.unlink()
                concat_file.unlink()
                concat_out.unlink()
                temp_dir.rmdir()
                
                return output_path
    """)

                ctx.save("resolve", {k: v.model_dump() for k, v in resolver_results.items()})
                progress.advance(task)
                
                task = progress.add_task("🎲 RL Bandit Selecting...", total=1)
                bandit = RLBandit(novelty_weight=0.15)
                bandit_decisions, initial_irs = {}, {}
                for intent in intents:
                    dec = bandit.choose(intent, resolver_results[intent.id], song_ast, list(initial_irs.values()))
                    bandit_decisions[intent.id] = dec
                    resolver.mark_clip_used(dec.chosen_candidate.clip_dna_id)
                    chosen_id = dec.chosen_candidate.clip_dna_id
                    matching_clip = next((c for c in clip_dna if c.id == chosen_id), None)
                    clip_path = matching_clip.file_path if matching_clip else "unknown.mp4"
                    initial_irs[intent.id] = RenderIR(
                        clip_file_path=clip_path,
                        trim_in_ms=dec.chosen_candidate.trim_start_ms,
                        trim_out_ms=dec.chosen_candidate.trim_end_ms,
                        main_audio_track_id=song_ast.id, source_intent_id=intent.id, replay_log_id="")
                    if matching_clip:
                        for segment in matching_clip.segments:
                            if segment.start_ms == dec.chosen_candidate.trim_start_ms:
                                segment_id = f"{chosen_id}_{segment.segment_id}"
                                resolver.mark_segment_used(segment_id, chosen_id, intent.id, dec.reward_prediction)
                                break
                progress.advance(task)
                
                task = progress.add_task("⚙️ Running Optimizer Passes...", total=1)
                clip_dna_map = {c.id: c for c in clip_dna}
                optimizer = OptimizerPipeline(song_ast, clip_dna_map)
                timeline = optimizer.optimize_all(intents, resolver_results, bandit_decisions, initial_irs)
                ctx.save("optimize", {"timeline": [t.model_dump() for t in timeline], "logs": [l.model_dump() for l in optimizer.replay_logs]})
                progress.advance(task)
                
                task = progress.add_task("🎬 Rendering Final Video...", total=1)
                out_dir = Path.cwd() / "output"
                renamer = SmartRenamer(out_dir)
                base_name = f"{song_ast.artist}_{song_ast.title}"
                out_file = renamer.get_next_version(base_name, platform, ".mp4")
                renderer = FFmpegBackend(platform_config)
                try:
                    renderer.render(timeline, out_file, mp3)
                    if out_file.exists() and out_file.stat().st_size > 0:
                        render_success = True
                    else:
                        console.print("[bold red]❌ Render produced empty or missing file.[/bold red]")
                except Exception as e:
                    console.print(f"[bold red]❌ Render failed: {e}[/bold red]")
                progress.advance(task)
                
                task = progress.add_task("📊 Generating Report...", total=1)
                reporter = ReportGenerator(song_ast, optimizer.replay_logs)
                reporter.generate_html(Path.cwd() / "reports" / "render_report.html")
                ReplayLogger().save_logs(optimizer.replay_logs)
                progress.advance(task)

            if render_success:
                console.print("\\n[bold green]🎉 Compilation Successful![/bold green]")
                console.print(f"📁 Video: {out_file}")
                console.print(f"📊 Report: ./reports/render_report.html")
            else:
                console.print("\\n[bold red]❌ Compilation Failed during rendering.[/bold red]")
                console.print("Check the FFmpeg error messages above for details.")
            
            diversity_stats = resolver.get_diversity_statistics()
            console.print(f"\\n[bold cyan]🎯 Diversity Statistics:[/bold cyan]")
            console.print(f"   • Total segments: {diversity_stats['total_segments']}")
            console.print(f"   • Used once: {diversity_stats['used_once']}")
            console.print(f"   • Used multiple: {diversity_stats['used_multiple']}")
            
            total_duration_ms = sum(ir.trim_out_ms - ir.trim_in_ms for ir in timeline)
            console.print(f"\\n[bold cyan]📊 Timeline Statistics:[/bold cyan]")
            console.print(f"   • Total shots: {len(timeline)}")
            console.print(f"   • Total duration: {total_duration_ms / 1000:.1f}s")
            if song_ast:
                console.print(f"   • Song duration: {song_ast.total_duration_ms / 1000:.1f}s")

        @app.command()
        def replay(shot_id: str = typer.Argument(...)):
            logs = ReplayLogger().load_all()
            log = next((l for l in logs if l.shot_id == shot_id), None)
            if log:
                console.print(f"\\n[bold cyan]📋 Intent:[/bold cyan] {log.intent.song_section} | Energy: {log.intent.need_energy}")
                console.print(f"[bold green]✅ Chosen:[/bold green] {log.bandit_decision.chosen_candidate.clip_dna_id} (Reward: {log.final_reward:.2f})")
                console.print(f"[bold magenta]⚙️ Passes:[/bold magenta] {', '.join(log.optimizer_passes_applied)}")
            else:
                console.print(f"[red]Shot {shot_id} not found.[/red]")

        if __name__ == "__main__":
            app()
    """)
