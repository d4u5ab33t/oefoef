import os



import time



from playwright.sync_api import sync_playwright







class SynapsePostAgent:



    def __init__(self, video_path, caption, account_cookie_dir):



        self.video_path = os.path.abspath(video_path)



        self.caption = caption



        self.cookie_dir = account_cookie_dir







    def deploy_to_tiktok(self):



        print(f"[AGENT] Starte Posting-Vorgang für Node @la.bat.ne.da...")



        



        with sync_playwright() as p:



            # Startet einen lokalen Browser, der deine echten Login-Daten (Cookies) nutzt



            browser = p.chromium.launch_persistent_context(



                user_data_dir=self.cookie_dir,



                headless=False, # Zeigt das Fenster, um Captchas manuell zu lösen, falls nötig



                args=["--disable-blink-features=AutomationControlled"]



            )



            



            page = browser.new_page()



            page.goto("https://www.tiktok.com/creator-center/upload?lang=de-DE")



            page.wait_for_load_state("networkidle")



            



            print("[AGENT] Infiltriere Upload-Maske...")



            # Lokalisierung des Datei-Inputs über das lokale System



            file_input = page.locator('input[type="file"]')



            file_input.set_files(self.video_path)



            print("[AGENT] Video-Upload läuft im Hintergrund...")



            



            # Wartezeit für das Rendering und den Upload im Web-Interface



            time.sleep(15) 



            



            print("[AGENT] Injiziere optimierte SEO-Caption...")



            # Findet das Textfeld für die Videobeschreibung



            caption_box = page.locator('div[contenteditable="true"]').first



            caption_box.fill(self.caption)



            



            print("[AGENT] Warte auf finalen Release-Befehl...")



            time.sleep(5)



            



            # Klickt autonom auf den "Posten"-Button



            # page.locator('button:has-text("Posten")').click() 



            



            print("[AGENT] ERFOLG: Clip wurde in die TikTok-Infrastruktur geschleust.")



            browser.close()

import os
import time
from playwright.sync_api import sync_playwright

class SynapsePostAgent:
    def __init__(self, video_path, caption, account_cookie_dir, platform="tiktok"):
        self.video_path = os.path.abspath(video_path)
        self.caption = caption
        self.cookie_dir = account_cookie_dir
        self.platform = platform

    def deploy_to_platform(self):
        print(f"[AGENT] Starte Posting für {self.platform}...")

        with sync_playwright() as p:
            # Browser mit deinen lokalen Cookies starten (keine Cloud!)
            browser = p.chromium.launch_persistent_context(
                user_data_dir=self.cookie_dir,
                headless=False,  # Sichtbar für Captcha-Lösung
                args=["--disable-blink-features=AutomationControlled"]
            )
            page = browser.new_page()

            # Plattform-spezifische URLs
            if self.platform == "tiktok":
                page.goto("https://www.tiktok.com/creator-center/upload?lang=de-DE")
            elif self.platform == "instagram":
                page.goto("https://www.instagram.com/accounts/emails_and_sms/")
                # (Hier: Upload-Logik für Reels einfügen)
            elif self.platform == "youtube":
                page.goto("https://www.youtube.com/upload")

            page.wait_for_load_state("networkidle")
            print(f"[AGENT] Infiltriere {self.platform}-Upload-Maske...")

            # Video hochladen
            file_input = page.locator('input[type="file"]')
            file_input.set_files(self.video_path)
            print("[AGENT] Video-Upload läuft...")
            time.sleep(15)  # Warte auf Upload

            # Caption einfügen
            if self.platform == "tiktok":
                caption_box = page.locator('div[contenteditable="true"]').first
                caption_box.fill(self.caption)
                print("[AGENT] Caption injiziert...")
            elif self.platform == "instagram":
                # Reels-Caption-Logik hier
                pass
            elif self.platform == "youtube":
                # YouTube-Beschreibung hier
                pass

            time.sleep(5)
            print(f"[AGENT] ERFOLG: Clip wurde auf {self.platform} geschleust!")
            browser.close()

if __name__ == "__main__":
    import argparse
    parser = argparse.ArgumentParser()
    parser.add_argument("--video", type=str, required=True, help="Pfad zum Video")
